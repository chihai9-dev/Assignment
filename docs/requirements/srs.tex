\section{Giới thiệu tổng quan (Introduction)}
Tài liệu này xác định danh mục \textbf{Yêu cầu chức năng (Functional Requirements -- FR)} và \textbf{Yêu cầu phi chức năng (Non-Functional Requirements -- NFR)} cho hệ thống đặt đồ uống trực tuyến BrewLite. Mọi yêu cầu được phân loại mức độ ưu tiên theo mô hình \textbf{MoSCoW} (\textit{Must have, Should have, Could have, Won't have}), đi kèm các \textbf{Tiêu chí chấp nhận (Acceptance Criteria)} đo lường và kiểm chứng được, làm cơ sở truy vết thiết kế kiến trúc, hiện thực mã nguồn và xây dựng kịch bản kiểm thử tự động.

\section{Danh sách Yêu cầu Chức năng (Functional Requirements -- FR)}

Mức độ ưu tiên phân loại theo quy chuẩn \textbf{MoSCoW}:
\begin{itemize}[itemsep=1pt]
    \item \textbf{Must have (M):} Bắt buộc phải có để hệ thống hoạt động trong phạm vi MVP (Sprint 1 và Sprint 2).
    \item \textbf{Should have (S):} Cần thiết để trải nghiệm người dùng hoàn chỉnh.
    \item \textbf{Could have (C):} Tính năng bổ trợ nếu có thêm thời gian (Gợi ý món hot).
    \item \textbf{Won't have (W):} Tạm hoãn, không triển khai trong phạm vi MVP (Thanh toán tiền mặt).
\end{itemize}

\hline
\textbf{FR-03} & \textbf{Giỏ hàng} & Quản lý giỏ hàng cục bộ & Thêm món vào giỏ, tăng/giảm số lượng, xóa món và xem tổng tiền. & \textbf{Must} & 
\textbf{AC-03.1:} Giỏ lưu đúng: \texttt{productId, size, qty, toppings[]}.\newline
\textbf{AC-03.2:} Chỉnh sửa số lượng ($1 \le qty \le 99$) hoặc xóa món.\newline
\textbf{AC-03.3:} Dữ liệu giỏ hàng được duy trì sau khi tải lại trang (reload).\newline
\textbf{AC-03.4:} Tổng tiền tính đúng: $\sum ((\text{Giá size} + \sum \text{Topping}) \times \text{Số lượng})$. \\
\hline
\textbf{FR-04} & \textbf{Xác thực} & Đăng ký \& Đăng nhập & Đăng ký Email/Mật khẩu và đăng nhập nhận JWT Token xác thực. & \textbf{Must} & 
\textbf{AC-04.1:} Mật khẩu mã hóa một chiều bằng \texttt{bcrypt}.\newline
\textbf{AC-04.2:} Từ chối nếu Email đã tồn tại trong hệ thống.\newline
\textbf{AC-04.3:} Đăng nhập thành công trả về JWT Token chứa \texttt{userId} và \texttt{role} (\texttt{customer} hoặc \texttt{barista}). \\
\hline

\textbf{AC-05.1:} API \texttt{POST /orders} yêu cầu JWT Token.\newline
\textbf{AC-05.2:} Tạo Order, OrderItem, OrderItemTopping và trừ kho phải nằm trong một Database Transaction (Atomic).\newline
\textbf{AC-05.3:} Khóa lạc quan qua cột \texttt{version}. Báo lỗi 400 nếu không đủ hàng tồn kho. \\
\hline
\textbf{FR-06} & \textbf{Thanh toán} & Thanh toán đơn hàng & Khách hàng thanh toán đơn \texttt{PENDING} qua Ví điện tử hoặc Thẻ mô phỏng. & \textbf{Must} & 
\textbf{AC-06.1:} API \texttt{POST /payments} nhận \texttt{orderId}, \texttt{method}, \texttt{amount}, \texttt{idempotencyKey}.\newline
\textbf{AC-06.2:} Trùng \texttt{idempotencyKey} trả về kết quả cũ, không trừ tiền hay tạo giao dịch lần 2.\newline
\textbf{AC-06.3:} Thành công chuyển đơn sang \texttt{PAID}. Thất bại chuyển sang \texttt{PAYMENT\_FAILED}. \\
\hline
\textbf{FR-07} & \textbf{Lịch sử} & Xem đơn \& Tích điểm & Khách xem danh sách đơn đã mua và điểm tích lũy thành viên. & \textbf{Must} & 
\textbf{AC-07.1:} \texttt{GET /orders} lọc theo \texttt{userId} đang đăng nhập.\newline
\textbf{AC-07.2:} Đơn \texttt{PAID} tự động cộng điểm: $10.000\text{ VNĐ} = 1\text{ điểm}$ vào \texttt{loyaltyPoints}. \\
\hline
\textbf{FR-08} & \textbf{Tiến độ đơn} & Cập nhật pha chế & Barista xem đơn \texttt{PAID} và cập nhật lần lượt sang \texttt{PREPARING}, \texttt{READY}, \texttt{COMPLETED}. & \textbf{Must} & 
\textbf{AC-08.1:} API \texttt{PATCH /orders/:id/status} yêu cầu \texttt{role = barista}.\newline
\textbf{AC-08.2:} Chuyển trạng thái theo đúng máy trạng thái FSM. Không cho phép nhảy cóc trạng thái bất hợp lệ. \\
\hline
\textbf{FR-09} & \textbf{Quản lý món} & Bật/tắt món \& topping & Bật/tắt khả dụng của sản phẩm hoặc topping khi quán tạm hết nguyên liệu. & \textbf{Should} & 
\textbf{AC-09.1:} Cập nhật cờ \texttt{isAvailable} qua \texttt{PATCH /api/toppings/:id}.\newline
\textbf{AC-09.2:} Món/topping tắt khả dụng lập tức ẩn khỏi danh sách lựa chọn của khách hàng. \\
\hline
\textbf{FR-10} & \textbf{Dọn dẹp đơn} & Hủy đơn quá hạn tự động & Tự động hủy đơn hàng \texttt{PENDING} quá hạn và hoàn trả tồn kho. & \textbf{Must} & 
\textbf{AC-10.1:} Tự động hủy các đơn \texttt{PENDING} quá thời hạn (15 phút) sang \texttt{CANCELLED}.\newline
\textbf{AC-10.2:} Hoàn trả chính xác số lượng tồn kho (\textit{Stock Rollback}) cho sản phẩm tương ứng. \\
\hline
\textbf{FR-11} & \textbf{Ưu đãi TV} & Giảm giá theo hạng TV & Áp dụng chiết khấu dựa trên hạng thành viên tích lũy của khách hàng. & \textbf{Should} & 
\textbf{AC-11.1:} Tính toán hạng Bạc (5\%), Vàng (10\%), Kim Cương (15\%) dựa trên \texttt{loyaltyPoints}.\newline
\textbf{AC-11.2:} Tự động trừ tiền giảm giá vào tổng đơn hàng khi đặt hàng. \\
\hline
\textbf{FR-12} & \textbf{Gợi ý món} & Gợi ý đồ uống phổ biến & Hiển thị danh sách top đồ uống được đặt nhiều nhất tại trang chủ. & \textbf{Could} & 
\textbf{AC-12.1:} Hiển thị mục "Món bán chạy" (Top trending) tại màn hình Menu.\newline
\textbf{AC-12.2:} Cho phép người dùng chọn nhanh món từ danh mục gợi ý. \\
\hline
\textbf{FR-13} & \textbf{Tiền mặt} & Thanh toán tiền mặt tại quầy & Thanh toán bằng tiền mặt trực tiếp tại quầy thu ngân. & \textbf{Won't} & 
\textbf{AC-13.1:} Hệ thống hiển thị thông báo rõ ràng chỉ chấp nhận phương thức thanh toán trực tuyến mô phỏng trong phiên bản hiện tại. \\

\section{Máy trạng thái đơn hàng (Order State Machine)}
Vòng đời đơn hàng tuân thủ nghiêm ngặt máy trạng thái hữu hạn (\textit{Finite State Machine -- FSM}):

\begin{center}
\small
\fbox{\begin{minipage}{0.9\textwidth}
\centering
\vspace{0.2cm}
\textbf{Khách đặt đơn} $\xrightarrow{\text{POST /orders}}$ \texttt{PENDING} $\xrightarrow[\text{POST /payments}]{\text{Thanh toán thành công}}$ \texttt{PAID} $\xrightarrow{\text{Nhận pha chế}}$ \texttt{PREPARING} $\xrightarrow{\text{Pha chế xong}}$ \texttt{READY} $\xrightarrow{\text{Giao khách}}$ \texttt{COMPLETED}\\[0.2cm]
\textit{Các nhánh ngoại lệ:} \texttt{PENDING} $\xrightarrow{\text{Lỗi / Hết giờ}}$ \texttt{PAYMENT\_FAILED} \quad $\vert$ \quad \texttt{PENDING} $\xrightarrow{\text{Khách hủy}}$ \texttt{CANCELLED}
\vspace{0.2cm}
\end{minipage}}
\end{center}

\begin{itemize}[leftmargin=*,itemsep=1.5pt]
    \item \textbf{PENDING:} Đơn vừa tạo, chờ thanh toán (tồn kho đã được tạm giữ an toàn trong transaction).
    \item \textbf{PAID:} Đã thanh toán tiền thành công, hệ thống tự động cộng điểm \texttt{loyaltyPoints} cho khách.
    \item \textbf{PREPARING:} Nhân viên Barista tiếp nhận đơn và thực hiện pha chế theo công thức.
    \item \textbf{READY:} Thức uống đã pha xong, sẵn sàng ở quầy chờ khách đến lấy.
    \item \textbf{COMPLETED:} Khách đã nhận đồ uống tại quầy, quy trình đặt hàng hoàn tất.
    \item \textbf{PAYMENT\_FAILED / CANCELLED:} Đơn hàng bị hủy bỏ; hệ thống tự động hoàn trả lại số lượng tồn kho (\textit{Rollback stock}).
\end{itemize}
